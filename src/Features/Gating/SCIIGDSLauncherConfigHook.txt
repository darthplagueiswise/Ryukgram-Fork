// ── Generic hook helper ───────────────────────────────────────────────────────
// Uses MSHookMessageEx(class, selector) — safe even with shared trampoline IMPs.
// Stores a weak cache to avoid double-hooking.
static NSMutableSet<NSString *> *gHooked;

typedef BOOL (*BoolNoArg_t)(id, SEL);
static NSMutableDictionary<NSString *, NSValue *> *gOrigs;

static void HookBoolGetter(Class cls, SEL sel, BOOL(^force)(void)) {
    if (!cls || !sel) return;
    Method m = class_getInstanceMethod(cls, sel);
    if (!m) return;
    char ret[4]={0}; method_getReturnType(m, ret, sizeof(ret));
    if (ret[0] != 'B' && ret[0] != 'c') return;
    if (method_getNumberOfArguments(m) != 2) return;

    NSString *key = [NSString stringWithFormat:@"%s#%s",
                     class_getName(cls), sel_getName(sel)];
    if ([gHooked containsObject:key]) return;

    BOOL(^capturedForce)(void) = [force copy];
    BoolNoArg_t origFn = (BoolNoArg_t)method_getImplementation(m);
    gOrigs[key] = [NSValue valueWithPointer:(const void *)origFn];

    IMP hook = imp_implementationWithBlock(^BOOL(id _self) {
        if (capturedForce()) return YES;
        BoolNoArg_t o = (BoolNoArg_t)[gOrigs[key] pointerValue];
        return o ? o(_self, sel) : NO;
    });
    MSHookMessageEx(cls, sel, hook, nil);
    [gHooked addObject:key];
    gHookCount++;
}

// ── Install ───────────────────────────────────────────────────────────────────
static BOOL sInstalled = NO;

static void IGDSInstall(void) {
    IGDSReadPrefs();
    if (!sAll && !sLG && !sPrism) return;
    if (!gHooked) gHooked = [NSMutableSet set];
    if (!gOrigs)  gOrigs  = [NSMutableDictionary dictionary];

    Class cls = NSClassFromString(@"IGDSLauncherConfig");
    if (!cls) {
        IGDS_LOG("IGDSLauncherConfig not found yet");
        return;
    }

    for (NSString *sel in IGDSLiquidGlassPositive())
        HookBoolGetter(cls, NSSelectorFromString(sel), ^{ return (BOOL)(sAll || sLG); });

    for (NSString *sel in IGDSPrismPositive())
        HookBoolGetter(cls, NSSelectorFromString(sel), ^{ return (BOOL)(sAll || sPrism); });

    for (NSString *sel in IGDSGeneralPositive())
        HookBoolGetter(cls, NSSelectorFromString(sel), ^{ return sAll; });

    // Wordmark: each selector reads only C static cache inside the replacement block.
    HookBoolGetter(cls, NSSelectorFromString(@"isIGWordmark1aEnabled"), ^{ return sWM1a; });
    HookBoolGetter(cls, NSSelectorFromString(@"isIGWordmark1aAltEnabled"), ^{ return sWM1aAlt; });
    HookBoolGetter(cls, NSSelectorFromString(@"isIGWordmark1bEnabled"), ^{ return sWM1b; });
    HookBoolGetter(cls, NSSelectorFromString(@"isIGWordmark1bAltEnabled"), ^{ return sWM1bAlt; });

    sInstalled = gHookCount > 0;
    IGDS_LOG("installed=%d hooks=%lu LG=%d Prism=%d All=%d",
             (int)sInstalled, (unsigned long)gHookCount,
             (int)sLG, (int)sPrism, (int)sAll);
}

void SCIIGDSEnsureHooksInstalled(void) { IGDSInstall(); }

%ctor {
    @autoreleasepool {
        IGDSInstall();
        double d[] = {1.0, 3.0, 6.0};
        for (NSUInteger i = 0; i < 3; i++)
            dispatch_after(dispatch_time(DISPATCH_TIME_NOW,(int64_t)(d[i]*NSEC_PER_SEC)),
                           dispatch_get_main_queue(), ^{ IGDSInstall(); });
    }
}
